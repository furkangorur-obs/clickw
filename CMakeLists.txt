cmake_minimum_required(VERSION 3.20)
project(clickw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KF6GlobalAccel REQUIRED)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EVDEV REQUIRED IMPORTED_TARGET libevdev)
add_executable(clickw main.cpp)
target_link_libraries(clickw PRIVATE Qt6::Widgets Qt6::DBus KF6::GlobalAccel LayerShellQt::Interface PkgConfig::EVDEV)
target_compile_options(clickw PRIVATE -Wall -Wextra)
configure_file(ClickW.desktop.in ClickW.desktop @ONLY)
